class Nivel3 extends Phaser.Scene { constructor() { super({ key: 'Nivel3' }); } init() { this.vidas = 3; this.pedido = { sillas: 0, cafes: 0 }; this.objetivo = { sillas: 5, cafes: 3 }; } preload() { // Carga de imágenes desde la raíz this.load.image('plaza', 'plaza.png'); this.load.image('boss', 'boss.png'); this.load.image('silla', 'silla.png'); this.load.image('cafe', 'cafe.png'); this.load.image('barril', 'barrilexp.png'); this.load.image('pj_normal', 'personajeh.png'); this.load.image('pj_muerto', 'personaje.png'); for (let i = 1; i <= 5; i++) { this.load.image(`e${i}`, `e${i}.png`); } } create() { // 1. Fondo Plaza (Nivel 3) this.add.image(400, 600, 'plaza').setDisplaySize(800, 1200); // 2. Boss: Ajustado en el balcón izquierdo // Bajamos la escala porque el jefe es muy grande en el archivo original this.boss = this.add.image(110, 310, 'boss').setScale(0.12); // 3. Jugador: Contenedor const pjSprite = this.add.sprite(0, 0, 'pj_normal').setScale(0.12); const txtNombre = this.add.text(0, -80, "OPERARIO", { fontSize: '20px', fill: '#0ff', fontStyle: 'bold', stroke: '#000', strokeThickness: 3 }).setOrigin(0.5); this.player = this.add.container(400, 1080, [pjSprite, txtNombre]); this.physics.world.enable(this.player); this.player.body.setCollideWorldBounds(true); // Hitbox ajustada para que sea justo el cuerpo del operario this.player.body.setSize(80, 120); this.objetos = this.physics.add.group(); // 4. UI: Movida un poco para que el Boss no la tape this.ui = this.add.text(250, 20, '', { fontSize: '22px', fill: '#fff', backgroundColor: '#000000b3', padding: 8 }); this.actualizarUI(); // Lanzamiento cada 2.5 segundos (un poco más lento para dar tiempo) this.time.addEvent({ delay: 2500, callback: this.lanzar, callbackScope: this, loop: true }); // Movimiento lateral this.input.on('pointermove', p => { this.player.x = Phaser.Math.Clamp(p.x, 60, 740); }); this.physics.add.overlap(this.player, this.objetos, this.recolectar, null, this); } lanzar() { const items = ['silla', 'cafe', 'barril']; const item = Phaser.Utils.Array.GetRandom(items); // El proyectil sale desde la posición del Boss const obj = this.objetos.create(this.boss.x + 40, this.boss.y, item); // CORRECCIÓN DE ESCALAS: // Si es café o silla, escala muy pequeña (0.05 a 0.08) if (item === 'cafe') obj.setScale(0.05); else if (item === 'silla') obj.setScale(0.08); else obj.setScale(0.1); // Barril un poco más grande obj.body.setGravityY(700); // Impulso parabólico hacia la derecha y arriba obj.body.setVelocity(Phaser.Math.Between(150, 400), Phaser.Math.Between(-500, -700)); obj.setData('tipo', item); // Rotación para que parezca que vuelan por el aire obj.setAngularVelocity(Phaser.Math.Between(-100, 100)); } recolectar(p, o) { const tipo = o.getData('tipo'); o.destroy(); if (tipo === 'barril') { this.vidas--; this.cameras.main.shake(150, 0.02); if (this.vidas <= 0) this.morir(); } else { if (tipo === 'silla') this.pedido.sillas++; if (tipo === 'cafe') this.pedido.cafes++; this.actualizarUI(); this.verificarVictoria(); } } actualizarUI() { this.ui.setText(`PLAZA - Sillas: ${this.pedido.sillas}/5 | Cafés: ${this.pedido.cafes}/3 | ❤️: ${this.vidas}`); } verificarVictoria() { if (this.pedido.sillas >= this.objetivo.sillas && this.pedido.cafes >= this.objetivo.cafes) { this.add.text(400, 600, '¡PEDIDO LISTO!', { fontSize: '50px', fill: '#0f0', stroke: '#000', strokeThickness: 6 }).setOrigin(0.5); this.physics.pause(); } } morir() { this.physics.pause(); this.player.list[0].setTexture('pj_muerto'); this.add.text(400, 600, 'GAME OVER', { fontSize: '60px', fill: '#f00', stroke: '#000', strokeThickness: 6 }).setOrigin(0.5); } update() { // Limpiar objetos que caen al vacío this.objetos.getChildren().forEach(obj => { if (obj.y > 1250) obj.destroy(); }); } }
