/** * WATERWORLD - Estructura Limpia * Personaje: Recorte manual para evitar errores de hoja irregular. */ const config = { type: Phaser.AUTO, width: 800, height: 1200, parent: 'game-container', physics: { default: 'arcade', arcade: { gravity: { y: 0 } } }, scene: [Boot, MainMenu, Game] }; const game = new Phaser.Game(config); class Boot extends Phaser.Scene { constructor() { super('Boot'); } preload() { // Carga de imágenes this.load.image('fondo', 'fondo.jpg'); // Tu archivo de fondo this.load.image('personajes', 'personaje.jpg'); // Tu hoja de sprites } create() { // Definimos las áreas de recorte manualmente (Coordenadas aproximadas según tu imagen) // [X, Y, Ancho, Alto] this.textures.addSpriteSheetFromLayer('player_m', 'personajes', { x: 0, y: 0, width: 200, height: 250 }); // Oficinista Azul this.textures.addSpriteSheetFromLayer('player_f', 'personajes', { x: 0, y: 250, width: 200, height: 250 }); // Oficinista Rosa this.scene.start('MainMenu'); } } class MainMenu extends Phaser.Scene { constructor() { super('MainMenu'); } create() { this.add.image(400, 600, 'fondo').setDisplaySize(800, 1200).setAlpha(0.6); this.add.text(400, 300, 'WATERWORLD', { fontSize: '80px', fill: '#0ff', fontStyle: 'bold' }).setOrigin(0.5); this.add.text(400, 500, 'SELECCIONA PERSONAJE', { fontSize: '30px', fill: '#fff' }).setOrigin(0.5); // Botón Hombre let btnM = this.add.image(300, 700, 'personajes', 0).setInteractive().setScale(0.8); btnM.on('pointerdown', () => this.iniciarJuego('player_m')); // Botón Mujer let btnF = this.add.image(500, 700, 'personajes', 4).setInteractive().setScale(0.8); btnF.on('pointerdown', () => this.iniciarJuego('player_f')); } iniciarJuego(textura) { this.registry.set('p_textura', textura); this.scene.start('Game'); } } class Game extends Phaser.Scene { constructor() { super('Game'); } create() { this.add.image(400, 600, 'fondo').setDisplaySize(800, 1200); // Recuperamos la textura elegida const texturaElegida = this.registry.get('p_textura'); // Creamos al jugador this.player = this.physics.add.sprite(400, 900, 'personajes'); // Aquí forzamos el cuadro inicial para que no se vea toda la hoja this.player.setFrame(texturaElegida === 'player_m' ? 0 : 4); // --- ANIMACIÓN PROCEDURAL (El Latido que pediste) --- this.tweens.add({ targets: this.player, scaleX: 1.1, scaleY: 0.9, duration: 600, yoyo: true, repeat: -1, ease: 'Sine.easeInOut' }); this.cursors = this.input.keyboard.createCursorKeys(); } update() { this.player.setVelocity(0); if (this.cursors.left.isDown) this.player.setVelocityX(-300); if (this.cursors.right.isDown) this.player.setVelocityX(300); if (this.cursors.up.isDown) this.player.setVelocityY(-300); if (this.cursors.down.isDown) this.player.setVelocityY(300); } }
